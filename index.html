// Procura o código exato e também equivalências comuns entre UPC-A (12 dígitos) // e EAN-13 (o mesmo código com zero à esquerda). function extrairDescricaoProduto(valor) { if (!valor) return ""; if (typeof valor === "object") { if (typeof valor.produto === "string" && valor.produto.trim()) { return valor.produto.trim(); } if (typeof valor.descricao === "string" && valor.descricao.trim()) { return valor.descricao.trim(); } return ""; } return String(valor).trim(); } function encontrarProdutoNaBase(base, codigo) { const candidatos = [String(codigo || "").trim()]; const original = candidatos[0]; if (/^\d{12}$/.test(original)) candidatos.push("0" + original); if (/^0\d{12}$/.test(original)) candidatos.push(original.slice(1)); for (const candidato of candidatos) { if (Object.prototype.hasOwnProperty.call(base, candidato)) { const descricao = extrairDescricaoProduto(base[candidato]); if (descricao) { return { codigoEncontrado: candidato, descricao }; } } } return null; } // Consulta EAN exclusivamente no JSON; não consulta a planilha para descobrir descrições. function buscarDescricaoProduto(codigoInformado) { const codigo = String(codigoInformado || "").trim(); const sequencia = ++consultaSequencia; if (!/^\d{8,14}$/.test(codigo)) { $("descricaoProduto").style.display = "none"; return; } const emCache = lerCache(codigo); if (emCache && emCache.status === "found") { mostrarDescricao("Produto encontrado na base JSON.", "success", emCache.descricao); registrarNomeAtual(codigo, emCache.descricao); return; } mostrarDescricao("Consultando a base JSON de produtos..."); carregarProdutosJson().then(base => { if (sequencia !== consultaSequencia) return; const produto = encontrarProdutoNaBase(base, codigo); if (produto) { const descricao = produto.descricao; gravarCache(codigo, { status: "found", descricao }); mostrarDescricao("Produto encontrado na base JSON.", "success", descricao); registrarNomeAtual(codigo, descricao); } else { gravarCache(codigo, { status: "not_found" }); mostrarDescricao("EAN não encontrado no JSON. O lançamento exigirá confirmação.", "error"); registrarNomeAtual(codigo, "Não cadastrado (" + codigo + ")"); } }).catch(() => { if (sequencia !== consultaSequencia) return; mostrarDescricao("Não foi possível carregar produtos_ean.json. O lançamento ficará bloqueado até a base ser carregada.", "error"); }); }
